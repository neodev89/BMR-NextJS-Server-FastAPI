# app/repositories/user_repo.py
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select
from models_db.user_model import UserORM  # Il tuo modello SQLAlchemy
from app.schemas.user_schema import UserCreate  # Il tuo schema Pydantic per la creazione

class UserRepository:
    def __init__(self, db: AsyncSession):
        self.db = db

    async def get_by_id(self, user_id: int) -> UserORM | None:
        result = await self.db.execute(select(UserORM).where(UserORM.id == user_id))
        return result.scalar_one_or_none()

    async def create(self, user_data: UserCreate) -> UserORM:
        db_user = UserORM(**user_data.model_dump())
        self.db.add(db_user)
        await self.db.commit()
        await self.db.refresh(db_user)
        return db_user
